server {
    server_name mrpapshop.com www.mrpapshop.com;

    # Límite de body para uploads de imágenes (HEIC/PNG/JPG pueden pesar hasta 20 MB).
    # El backend Express también valida el tamaño; este límite debe ser >= al del backend.
    client_max_body_size 25M;

    location / {
        proxy_pass http://127.0.0.1:3000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        # IP real del visitante y esquema original — sin esto, Next.js nunca ve
        # la IP pública real (necesaria para la detección de país/idioma) y
        # todo llega como si viniera de 127.0.0.1.
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_cache_bypass $http_upgrade;
        # Timeouts generosos para uploads lentos (conexiones móviles).
        proxy_read_timeout 120s;
        proxy_send_timeout 120s;
    }

    listen 443 ssl; # managed by Certbot
    ssl_certificate /etc/letsencrypt/live/mrpapshop.com/fullchain.pem; # managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/mrpapshop.com/privkey.pem; # managed by Certbot
    include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot

}
server {
    if ($host = mrpapshop.com) {
        return 301 https://$host$request_uri;
    } # managed by Certbot


    listen 80;
    server_name mrpapshop.com www.mrpapshop.com;
    return 404; # managed by Certbot


}
